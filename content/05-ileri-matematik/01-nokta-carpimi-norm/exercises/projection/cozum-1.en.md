$$
\frac{\mathbf{a} \cdot \mathbf{b}}{\|\mathbf{b}\|} = \frac{26}{5} = 5.2
$$

$$
\frac{\mathbf{a} \cdot \mathbf{b}}{\mathbf{b} \cdot \mathbf{b}}\,\mathbf{b} = \frac{26}{25}\,(3,\ 4) = (3.12,\ 4.16)
